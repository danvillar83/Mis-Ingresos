<script>

  // ---- Storage helpers ----

  const keyFor = (ym) => `ingresos_app_${ym}`;

  const extrasKeyFor = (ym) => `ingresos_app_extras_${ym}`;

  const fmtCLP = (n) => (Number(n)||0).toLocaleString("es-CL");

  function todayISO(){

    const d = new Date();

    const pad = (x)=> String(x).padStart(2,"0");

    return `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;

  }

  function monthISO(){

    const d = new Date();

    const pad = (x)=> String(x).padStart(2,"0");

    return `${d.getFullYear()}-${pad(d.getMonth()+1)}`;

  }

  function getMonth(){

    return document.getElementById("month").value || monthISO();

  }

  function getRate(){

    return Number(document.getElementById("rate").value || 1200);

  }

  function loadRows(ym){

    return JSON.parse(localStorage.getItem(keyFor(ym)) || "[]");

  }

  function saveRows(ym, rows){

    localStorage.setItem(keyFor(ym), JSON.stringify(rows));

  }

  function loadExtras(ym){

    return JSON.parse(localStorage.getItem(extrasKeyFor(ym)) || "[]");

  }

  function saveExtras(ym, extras){

    localStorage.setItem(extrasKeyFor(ym), JSON.stringify(extras));

  }

  // ---- Actions ----

  function addDay(){

    const ym = getMonth();

    const rows = loadRows(ym);

    const date = document.getElementById("date").value;

    if(!date){ alert("Selecciona una fecha"); return; }

    const packages = Number(document.getElementById("packages").value || 0);

    const withdrawals = Number(document.getElementById("withdrawals").value || 0);

    const uber = Number(document.getElementById("uber").value || 0);

    rows.push({ date, packages, withdrawals, uber });

    rows.sort((a,b)=> a.date.localeCompare(b.date));

    saveRows(ym, rows);

    // opcional: recordar último retiro

    localStorage.setItem("last_withdrawals_value", String(withdrawals));

    document.getElementById("packages").value = "";

    document.getElementById("uber").value = "";

    render();

  }

  function deleteRow(idx){

    const ym = getMonth();

    const rows = loadRows(ym);

    rows.splice(idx,1);

    saveRows(ym, rows);

    render();

  }

  function addExtra(){

    const ym = getMonth();

    const label = (document.getElementById("extraLabel").value || "").trim();

    const amount = Number(document.getElementById("extraAmount").value || 0);

    if(!label || amount<=0){

      alert("Completa nombre y monto del extra");

      return;

    }

    const extras = loadExtras(ym);

    extras.push({ label, amount });

    saveExtras(ym, extras);

    document.getElementById("extraLabel").value = "";

    document.getElementById("extraAmount").value = "";

    render();

  }

  function removeExtra(i){

    const ym = getMonth();

    const extras = loadExtras(ym);

    extras.splice(i,1);

    saveExtras(ym, extras);

    render();

  }

  function clearMonth(){

    const ym = getMonth();

    if(!confirm(`¿Seguro que quieres borrar TODO el mes ${ym}?`)) return;

    localStorage.removeItem(keyFor(ym));

    localStorage.removeItem(extrasKeyFor(ym));

    render();

  }

  function exportCSV(){

    const ym = getMonth();

    const rows = loadRows(ym);

    const extras = loadExtras(ym);

    const rate = getRate();

    let csv = "Mes,Fecha,CantPaquetes,Retiros,Uber,ValorPorPaquete\n";

    rows.forEach(r=>{

      csv += `${ym},${r.date},${r.packages},${r.withdrawals},${r.uber},${rate}\n`;

    });

    csv += "\nExtras\nNombre,Monto\n";

    extras.forEach(e=>{

      csv += `${e.label},${e.amount}\n`;

    });

    const blob = new Blob([csv], {type:"text/csv;charset=utf-8"});

    const url = URL.createObjectURL(blob);

    const a = document.createElement("a");

    a.href = url;

    a.download = `ingresos_${ym}.csv`;

    a.click();

    URL.revokeObjectURL(url);

  }

  // ---- Render ----

  function render(){

    const ym = getMonth();

    const rows = loadRows(ym);

    const extras = loadExtras(ym);

    const rate = getRate();

    // Table

    const tbody = document.getElementById("rows");

    tbody.innerHTML = "";

    rows.forEach((r,idx)=>{

      const tr = document.createElement("tr");

      tr.innerHTML = `

        <td>${r.date}</td>

        <td>${r.packages}</td>

        <td>$${fmtCLP(r.withdrawals)}</td>

        <td>$${fmtCLP(r.uber)}</td>

        <td><button class="danger" style="width:auto;padding:8px 10px;border-radius:10px;" onclick="deleteRow(${idx})">Eliminar</button></td>

      `;

      tbody.appendChild(tr);

    });

    // Extras view

    const extrasView = document.getElementById("extrasView");

    if(extras.length === 0){

      extrasView.innerHTML = `<span class="muted">Sin otros ingresos en este mes.</span>`;

    } else {

      extrasView.innerHTML = extras.map((e,i)=>`

        <div style="display:flex;justify-content:space-between;align-items:center;background:#f9fafb;border:1px solid #e5e7eb;padding:10px;border-radius:12px;margin-top:8px;">

          <div>${e.label}</div>

          <div><b>$${fmtCLP(e.amount)}</b>

            <button class="danger" style="width:auto;padding:6px 10px;border-radius:10px;margin-left:8px;" onclick="removeExtra(${i})">X</button>

          </div>

        </div>

      `).join("");

    }

    // KPIs (lo que pediste)

    const totalPackages = rows.reduce((s,r)=> s + (Number(r.packages)||0), 0);

    const totalByPackages = totalPackages * rate;

    const totalWithdrawals = rows.reduce((s,r)=> s + (Number(r.withdrawals)||0), 0);

    const totalUber = rows.reduce((s,r)=> s + (Number(r.uber)||0), 0);

    const totalExtras = extras.reduce((s,e)=> s + (Number(e.amount)||0), 0);

    const totalGeneral = totalByPackages + totalWithdrawals + totalExtras;

    document.getElementById("kpiPackages").innerText = `$${fmtCLP(totalByPackages)}`;

    document.getElementById("kpiWithdrawals").innerText = `$${fmtCLP(totalWithdrawals)}`;

    document.getElementById("kpiExtras").innerText = `$${fmtCLP(totalExtras)}`;

    document.getElementById("kpiTotal").innerText = `$${fmtCLP(totalGeneral)}`;

    // info pie

    const daysWorked = rows.filter(r => Number(r.packages) > 0).length;

    document.getElementById("footInfo").innerText =

      `Días trabajados (paq>0): ${daysWorked} | Total paquetes: ${totalPackages} | Total Uber (informativo): $${fmtCLP(totalUber)}`;

  }

  // Init defaults

  (function init(){

    document.getElementById("month").value = monthISO();

    document.getElementById("date").value = todayISO();

    // recordar el último retiro usado

    const lastW = localStorage.getItem("last_withdrawals_value");

    if (lastW) document.getElementById("withdrawals").value = lastW;

    document.getElementById("month").addEventListener("change", render);

    document.getElementById("rate").addEventListener("input", render);

    render();

  })();

</script>
